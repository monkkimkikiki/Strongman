cmake_minimum_required(VERSION 3.22.1)
project(bulknote_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp llama-build)
add_library(bulknote-ai SHARED bulknote_ai.cpp)
target_include_directories(bulknote-ai PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp/include ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp/ggml/include)
target_link_libraries(bulknote-ai PRIVATE llama android log)
